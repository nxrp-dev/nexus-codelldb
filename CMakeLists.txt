cmake_minimum_required(VERSION 3.21)
project(NexusCodeLLDB VERSION 1.12.3 LANGUAGES C CXX)

include(CTest)
include(cmake/CopyFiles.cmake)

# LLVM is supplied by the build environment and remains external at runtime.
set(LLVM_ROOT "" CACHE PATH "Installed LLVM root")
set(LLVM_SOURCE_DIR "" CACHE PATH "Matching LLVM sources, if LLDB headers are not installed")
set(LLDB_INCLUDE_DIRS "" CACHE STRING "LLDB include directories, including generated headers")
set(LLDB_LIBRARY "" CACHE FILEPATH "Installed liblldb shared library")
set(LLVM_TRIPLE "" CACHE STRING "Rust target triple (defaults to the build host)")
find_program(CARGO cargo REQUIRED)
find_program(RUSTC rustc REQUIRED)
if (NOT LLVM_TRIPLE)
    execute_process(COMMAND ${RUSTC} -vV OUTPUT_VARIABLE RustVersion COMMAND_ERROR_IS_FATAL ANY)
    string(REGEX MATCH "host: ([^\r\n]+)" RustHost "${RustVersion}")
    set(LLVM_TRIPLE "${CMAKE_MATCH_1}")
endif()

if (WIN32)
    set(ExeSuffix .exe)
    set(DylibSuffix .dll)
    set(PLATFORM_ID win32-x64 CACHE STRING "Package platform")
    string(APPEND CMAKE_CXX_FLAGS " /std:c++17")
elseif (APPLE)
    set(DylibSuffix .dylib)
    set(PLATFORM_ID darwin-${CMAKE_SYSTEM_PROCESSOR} CACHE STRING "Package platform")
    string(APPEND CMAKE_CXX_FLAGS " -std=c++17 -stdlib=libc++")
else()
    set(DylibSuffix .so)
    set(PLATFORM_ID linux-x64 CACHE STRING "Package platform")
    string(APPEND CMAKE_CXX_FLAGS " -std=c++17 -stdlib=libc++")
endif()

set(UpdateFile ${CMAKE_COMMAND} -E copy_if_different)
set(CargoFlags --locked --manifest-path=${CMAKE_SOURCE_DIR}/Cargo.toml --target=${LLVM_TRIPLE})
if (CMAKE_BUILD_TYPE MATCHES "Release|RelWithDebInfo|MinSizeRel")
    list(APPEND CargoFlags --release)
    set(CargoBuildType release)
else()
    set(CargoBuildType debug)
endif()

add_subdirectory(lldb)
if (WIN32)
    configure_file(cargo_config.windows.toml ${CMAKE_BINARY_DIR}/.cargo/config.toml @ONLY)
else()
    configure_file(cargo_config.unix.toml ${CMAKE_BINARY_DIR}/.cargo/config.toml @ONLY)
endif()
add_subdirectory(lang_support)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(adapter)

add_custom_target(check
    COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure
    DEPENDS adapter USES_TERMINAL)

# Explicit allowlist: no LLVM libraries, tools, Python runtime, or headers.
install(PROGRAMS ${CMAKE_BINARY_DIR}/adapter/codelldb${ExeSuffix} DESTINATION adapter)
install(PROGRAMS ${CMAKE_BINARY_DIR}/bin/codelldb-launch${ExeSuffix} DESTINATION bin)
if (WIN32)
    install(FILES ${MSDIA} DESTINATION adapter)
endif()
install(DIRECTORY adapter/scripts/ DESTINATION adapter/scripts FILES_MATCHING PATTERN "*.py")
install(DIRECTORY lang_support/ DESTINATION lang_support FILES_MATCHING PATTERN "*.py")
install(FILES LICENSE DESTINATION .)

set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME nexus-codelldb)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME nexus-codelldb-${PLATFORM_ID})
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
include(CPack)
